Siapkan Sheet Products (16 Kolom) let sheetProducts = ss.getSheetByName(SHEET_PRODUCTS_NAME); if (!sheetProducts) { sheetProducts = ss.insertSheet(SHEET_PRODUCTS_NAME); } if (sheetProducts.getLastRow() === 0) { const prodHeaders = [ 'id_produk', 'id_supplier', 'nama_supplier', 'kode_sku', 'nama_produk', 'kategori', 'satuan', 'harga_supplier', 'harga_jual_supplier', 'harga_supplier_lama', 'harga_jual_lama', 'total_stok', 'foto_url', 'link_download_foto', 'catatan', 'updated_at' ]; sheetProducts.getRange(1, 1, 1, prodHeaders.length).setValues([prodHeaders]); sheetProducts.getRange(1, 1, 1, prodHeaders.length).setFontWeight('bold').setBackground('#0f766e').setFontColor('#ffffff'); let mainFolder = null; const savedFolderId = extractGoogleId_(CUSTOM_DRIVE_FOLDER_ID_OR_URL) || scriptProps.getProperty('DRIVE_FOLDER_ID'); if (savedFolderId) { try { mainFolder = DriveApp.getFolderById(savedFolderId); } else { mainFolder = DriveApp.createFolder(MAIN_DRIVE_FOLDER_NAME); } } try { mainFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} scriptProps.setProperty('DRIVE_FOLDER_ID', mainFolder.getId()); const cleanId = String(idSupplier || 'SUP').trim(); const cleanName = String(namaSupplier || 'Supplier').trim().replace(/[\\/:*?"<>|]/g, ''); const desiredFolderName = cleanId + ' - ' + cleanName; const subFolders = mainFolder.getFolders(); while (subFolders.hasNext()) { const folder = subFolders.next(); const fName = folder.getName(); if (fName === desiredFolderName || fName.indexOf(cleanId + ' -') === 0) { if (fName !== desiredFolderName) { try { folder.setName(desiredFolderName); } catch (e) {} } return folder; } } const newSubFolder = mainFolder.createFolder(desiredFolderName); try { newSubFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} return newSubFolder; } /** * FUNGSI UTAMA INISIALISASI DATABASE & SUB-FOLDER GOOGLE DRIVE * Jalankan (Run) fungsi ini pertama kali di Editor Google Apps Script. */ function setupDatabase() { const db = setupSheetsOnly_(); const supName = String(supData[i][3] || '').trim(); const role = String(supData[i][4] || 'supplier').trim().toLowerCase(); if (supId && role !== 'admin') { const subFolder = getOrCreateSupplierSubFolder_(supId, supName); sheetSup.getRange(i + 1, 10).setValue(subFolder.getUrl()); subFolderCount++; } } Logger.log('======================================================'); Logger.log('SINKRONISASI DATABASE & SUB-FOLDER DRIVE BERHASIL!'); Logger.log('Spreadsheet URL : ' + db.ss.getUrl()); Logger.log('Folder Drive URL: ' + mainFolder.getUrl()); Logger.log('Total Sub-Folder: ' + subFolderCount + ' Sub-Folder Supplier'); Logger.log('======================================================'); return { spreadsheetId: db.ss.getId(), spreadsheetUrl: db.ss.getUrl(), mainDriveFolderId: mainFolder.getId(), mainDriveFolderUrl: mainFolder.getUrl(), subFolderCount: subFolderCount }; }
